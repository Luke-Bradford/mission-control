// @vitest-environment node
import {readFileSync,mkdirSync,writeFileSync} from "node:fs";
import {createHash} from "node:crypto";
import {expect,it} from "vitest";
import {browserSession,describeRendered} from "./browserRunner.ts";
import {desktopFixture} from "./desktopFixtures.ts";
const {renderAt}=browserSession();
const fixture=JSON.parse(readFileSync("/tmp/mc-1628-alert-fixtures.json","utf8"));
const out="/tmp/mc-1628-alert-evidence";mkdirSync(out,{recursive:true});
const receipts:unknown[]=[];
describeRendered("Held operator-message alert evidence",()=>{
 for(const phase of ["before","after"] as const) for(const theme of ["light","dark"] as const) it(`${phase} ${theme}`,async()=>{
  for(const [width,height] of [[1280,900],[1600,1000],[1600,700],[1920,940],[2560,1200]]){
   const api={...desktopFixture().api,"/api/attention":fixture.phases[phase].attention,"/api/attention/held":[],"/api/notifications":{open:[],handledToday:[]},"/api/flight":{seat:{id:"flight-1",call:"FLIGHT",model:"fixture",state:"idle",live:false},decisions:[]}};
   const page=await renderAt({page:"test/desktopBrowser.html",width,height,colorScheme:theme,search:"?path=/overview",api});
   try{
    await page.boxes("#page-overview");await page.evaluate(now=>{Date.now=()=>now},fixture.now);
    await page.click('a[href="/needs-you"]');await page.boxes("#page-needs .needs-grid");
    await page.evaluate(async()=>{await document.fonts.ready;await new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r)))},undefined);
    const facts=await page.evaluate(()=>({path:location.pathname,text:document.querySelector("#page-needs")?.textContent,detail:document.querySelector(".needs-detail")?.textContent,count:document.querySelector('[data-metric="needsYou"]')?.textContent,width:innerWidth,height:innerHeight,dpr:devicePixelRatio,scrollWidth:document.documentElement.scrollWidth,clock:Date.now(),rail:document.querySelector(".dockrail")!==null}),undefined);
    expect(facts.path).toBe("/needs-you");expect(facts.text).toContain("3 waiting");expect(facts.count).toBe("1");expect(facts.scrollWidth).toBeLessThanOrEqual(width);expect(page.errors()).toEqual([]);
    if(phase==="after"){expect(facts.detail).toContain("message is saved");expect(facts.detail).toContain("pending change");}else expect(facts.detail).toContain("usage stop line");
    const file=`held-alert-${phase}-${width}x${height}-${theme}.png`;await page.screenshot(out+"/"+file);
    receipts.push({phase,theme,width,height,file,sha256:createHash("sha256").update(readFileSync(out+"/"+file)).digest("hex"),capturedAt:new Date().toISOString(),arrival:"Overview -> Needs you navigation",scope:"viewport",...facts});
   }finally{await page.close()}
  }
  writeFileSync(out+"/capture.json",JSON.stringify({fixture:fixture.fixture,fixtureClock:fixture.now,receipts},null,2));
 },120000);
});
