import { readFileSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
import { join } from 'node:path';
import { pathToFileURL } from 'node:url';

const [beforeRoot, afterRoot, copiedDb, output, configPath] = process.argv.slice(2);
if (!configPath) throw new Error('Expected before-root, after-root, copied-db, output, config-path. Never use the live database.');
const hash = value => createHash('sha256').update(value).digest('hex');
const head = root => execFileSync('git', ['-C', root, 'rev-parse', 'HEAD'], { encoding: 'utf8' }).trim();
const storeFile = root => join(root, 'apps/server/src/store.ts');
const beforeHead = head(beforeRoot), afterHead = head(afterRoot);
const beforeStoreSha256 = hash(readFileSync(storeFile(beforeRoot)));
const afterStoreSha256 = hash(readFileSync(storeFile(afterRoot)));
const { Store: BeforeStore } = await import(pathToFileURL(storeFile(beforeRoot)).href);
const { Store: AfterStore } = await import(pathToFileURL(storeFile(afterRoot)).href);
const { loadConfig, configuredAgents } = await import(pathToFileURL(join(beforeRoot, 'apps/server/src/config.ts')).href);
const config = loadConfig(configPath);
const ids = [...new Set([...configuredAgents(config).map(agent => agent.id), 'CAPCOM'])];
const stores = { before: new BeforeStore(copiedDb), after: new AfterStore(copiedDb) };
const reads = { before: 0, after: 0 };
// Count executions, including reuse of a cached prepared statement. No production hook is added.
for (const [name, store] of Object.entries(stores)) {
  const prepared = store.statement.bind(store), wrapped = new WeakSet();
  store.statement = sql => {
    const statement = prepared(sql);
    if (sql.includes('FROM comms_mentions') && !wrapped.has(statement)) {
      const all = statement.all.bind(statement);
      statement.all = (...args) => { reads[name]++; return all(...args); };
      wrapped.add(statement);
    }
    return statement;
  };
}
const results = [];
try {
  for (let pair = 0; pair < 5; pair++) {
    for (const name of pair % 2 ? ['after', 'before'] : ['before', 'after']) {
      reads[name] = 0;
      const store = stores[name], started = performance.now();
      const value = ids.map(id => ({
        id,
        history: store.threadComms(id),
        afterRead: store.threadComms(id, store.threadReadCursors(id).get('comms')?.offset ?? 0),
      }));
      const elapsedMs = performance.now() - started, json = JSON.stringify(value);
      results.push({ pair, name, elapsedMs, threads: ids.length,
        rows: value.reduce((count, thread) => count + thread.history.length + thread.afterRead.length, 0),
        bytes: Buffer.byteLength(json), sha256: hash(json), actualMentionSelects: reads[name] });
    }
  }
  if (new Set(results.map(result => result.sha256)).size !== 1) throw new Error('Complete snapshot changed.');
  if (head(beforeRoot) !== beforeHead || head(afterRoot) !== afterHead ||
      hash(readFileSync(storeFile(beforeRoot))) !== beforeStoreSha256 ||
      hash(readFileSync(storeFile(afterRoot))) !== afterStoreSha256) throw new Error('Source moved during measurement.');
  const median = name => results.filter(result => result.name === name).map(result => result.elapsedMs).sort((a, b) => a - b)[2];
  const receipt = { at: new Date().toISOString(), beforeHead, afterHead, beforeStoreSha256, afterStoreSha256,
    method: 'Five alternating-order pairs, separate Store instances, one copied database. All configured agent threads plus CAPCOM, complete history and actual read-cursor variants. Full JSON must match. Native author checks running on the same host. This is not a whole-app idle acceptance observation.',
    medianMs: { before: median('before'), after: median('after') }, results };
  writeFileSync(output, JSON.stringify(receipt, null, 2) + '\n');
  console.log(JSON.stringify({ medianMs: receipt.medianMs, snapshot: results[0], output }));
} finally {
  stores.before.close();
  stores.after.close();
}
